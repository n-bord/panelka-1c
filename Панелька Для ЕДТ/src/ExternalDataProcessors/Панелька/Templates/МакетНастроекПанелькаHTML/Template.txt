function uiText(key, values) {
    var language = DATA.Ui && DATA.Ui.Language ? DATA.Ui.Language : "ru";
    var dictionary = Object.prototype.hasOwnProperty.call(UI_TEXT, language) ? UI_TEXT[language] : UI_TEXT.ru;
    var text = Object.prototype.hasOwnProperty.call(dictionary, key) ? dictionary[key] :
        Object.prototype.hasOwnProperty.call(UI_TEXT.ru, key) ? UI_TEXT.ru[key] : key;
    return text.replace(/\{([A-Za-z0-9_]+)\}/g, function (placeholder, name) {
        return values && Object.prototype.hasOwnProperty.call(values, name) ? String(values[name]) : placeholder;
    });
}
function localizeStaticTexts() {
    var language = DATA.Ui && DATA.Ui.Language ? DATA.Ui.Language : "ru";
    if (document.documentElement) document.documentElement.lang = Object.prototype.hasOwnProperty.call(UI_TEXT, language) ? language : "ru";
    var attributes = {"data-ui-text":"text", "data-ui-label":"aria-label", "data-ui-placeholder":"placeholder", "data-ui-title":"title"};
    for (var source in attributes) if (attributes.hasOwnProperty(source)) {
        var nodes = document.querySelectorAll('[' + source + ']');
        for (var i=0;i<nodes.length;i++) {
            if (!nodes[i].getAttribute) continue;
            var key = nodes[i].getAttribute(source);
            if (!key) continue;
            if (attributes[source] === "text") nodes[i].innerHTML = escapeHtml(uiText(key));
            else nodes[i].setAttribute(attributes[source], uiText(key));
        }
    }
    if (String(document.title || "").indexOf("PANELKASETTINGS") !== 0) document.title = uiText("settingsTitle");
}
function fieldLabel(id, key) {
    return '<label class="label" for="' + escapeHtml(id) + '">' + escapeHtml(uiText(key)) + '</label>';
}

function valueOr(value, fallback) {
    return value === null || typeof value === "undefined" ? fallback : value;
}

function escapeHtml(value) {
    var text = String(valueOr(value, ""));
    return text.replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/\"/g, "&quot;")
        .replace(/'/g, "&#39;");
}

function encode64(value) {
    var text = String(valueOr(value, ""));
    try {
        return window.btoa(unescape(encodeURIComponent(text)))
            .replace(/\+/g, "-")
            .replace(/\//g, "_")
            .replace(/=/g, ".");
    } catch (e) {
        return "";
    }
}

// Build a command for the title mailbox. Only publishBridgeCommand sends it to 1C.
function actionUrl(action, params) {
    var result = "panelkasettings://" + action;
    var pairs = [];
    var key;
    params = params || {};
    var page = params.page ? findPageByName(params.page) : null;
    var column = params.columnId ? findColumnByName(params.columnId) : (params.column ? findColumnByName(params.column, params.page) : null);
    if (page) params.pageId = page.Id;
    if (column) params.columnId = column.Id;
    if (action === 'choose_icon' || action === 'set_icon') {
        var object = params.kind === 'page' ? findPageByName(params.name) : findColumnByName(params.name);
        if (object) params.objectId = object.Id;
    }
    if(typeof params.index !== 'undefined' && !params.id && DATA.LinkIds) params.id=DATA.LinkIds[Number(params.index)]||'';
    for (key in params) {
        if (params.hasOwnProperty(key)) {
            pairs.push(key + "=" + encode64(params[key]));
        }
    }
    if (pairs.length > 0) {
        result += "?" + pairs.join("&");
    }
    return result;
}

function tryLocalSource(url) {
    if (bridgeBusy) return false;
    var action = url.split('?')[0].replace('panelkasettings://', '');
    var params = commandParameters(url);
    if (action !== 'set_metadata_group' && !(action === 'set_source_mode' && params.mode === 'metadata')) return false;
    updateEditorDraft();
    if (document.getElementById('panelkaEditorDraft').value) return false;
    var group = action === 'set_metadata_group' ? params.group : DATA.Ui.MetadataGroup;
    refreshMetadataCachePolicy();
    if (!metadataSourcesAllowed() || !metadataGroupAvailable(group)) return false;
    var cached = metadataSourceCache['group:' + group];
    if (!cached) return false;
    var started = clockMs(), state = captureViewState();
    DATA.Ui.SourceMode = 'metadata'; DATA.Ui.MetadataGroup = group; DATA.Sources = cached;
    DATA.Ui.State = JSON.stringify(state);
    renderSourceTabs(); renderMetadataGroups(); renderSourceList(); restoreViewState();
    recordPerformance({action:action, local:true, renderMs:clockMs()-started});
    return true;
}

function publishBridgeCommand(rawHref) {
    if (bridgeBusy) { showToast(uiText("wait")); return false; }
    var action = rawHref.split('?')[0].replace('panelkasettings://','');
    if (action === "save_close" && hasWebLinkDraft()) {
        showToast(uiText("finishWebDraft"));
        return false;
    }
    updateEditorDraft();
    var draft = document.getElementById('panelkaEditorDraft').value;
    if (action === 'open_additional_settings') additionalOpeningDraft = !!draft;
    if (draft && rawHref.split('?')[0] !== 'panelkasettings://apply_link_details') rawHref = appendCommandParameter(rawHref, 'draft', draft);
    rawHref = appendCommandParameter(rawHref, '__ui', JSON.stringify(captureViewState()));
    var page = findPageByName(DATA.Ui.ActivePage), column = findColumnByName(DATA.Ui.ActiveColumn);
    rawHref = appendCommandParameter(rawHref, '__activePageId', page ? page.Id : '');
    rawHref = appendCommandParameter(rawHref, '__activeColumnId', column ? column.Id : '');
    rawHref = appendCommandParameter(rawHref, '__selectedKind', DATA.Selected.Kind || 'None');
    rawHref = appendCommandParameter(rawHref, '__placementId', DATA.Selected.Placement ? DATA.Selected.Placement.Id : '');
    rawHref = appendCommandParameter(rawHref, '__sourceMode', DATA.Ui.SourceMode || '');
    rawHref = appendCommandParameter(rawHref, '__metadataGroup', DATA.Ui.MetadataGroup || '');
    pendingPerformance = {action:rawHref.split('?')[0].replace('panelkasettings://',''), started:clockMs()};
    bridgeBusy = true;
    activeBridgeAction = action;
    bridgeDisabledFields = [];
    var fields = document.querySelectorAll('input,textarea,select');
    for(var i=0;i<fields.length;i++) if(!fields[i].disabled && fields[i].id!=='panelkaEditorDraft' && fields[i].id!=='panelkaViewState') {
        bridgeDisabledFields.push(fields[i]);fields[i].disabled=true;
    }
    document.body.setAttribute('data-quick-command', /^(add_web|add_source|apply_link_details|apply_editor)$/.test(action) ? 'true' : 'false');
    document.body.setAttribute('aria-busy', 'true');
    bridgeSequence += 1;
    document.title = "PANELKASETTINGS|" + bridgeSequence + "|" + encode64(rawHref);
    return true;
}

function findAnchor(node) {
    while (node && node.tagName) {
        if (String(node.tagName).toUpperCase() === "A" || String(node.tagName).toUpperCase() === "BUTTON") {
            return node;
        }
        node = node.parentNode;
    }
    return null;
}

function getScrollTopById(id) {
    var node = document.getElementById(id);
    return node ? Math.max(0, parseInt(node.scrollTop || 0, 10) || 0) : 0;
}

function appendCommandParameter(url, name, value) {
    var separator = url.indexOf("?") >= 0 ? "&" : "?";
    return url + separator + name + "=" + encode64(String(value));
}

function appendScrollState(url) {
    url = appendCommandParameter(url, "__scrollCenter", getScrollTopById("previewShell"));
    url = appendCommandParameter(url, "__scrollSources", getScrollTopById("sourceList"));
    url = appendCommandParameter(url, "__scrollProperties", getScrollTopById("rightPane"));
    return url;
}

function handleBridgeClick(event) {
    event = event || window.event;

    // Если inline-обработчик уже отменил действие (например, пользователь
    // нажал «Отмена» в confirm), команду в 1С не отправляем.
    if (event && (event.defaultPrevented === true || event.returnValue === false)) {
        return;
    }

    var anchor = findAnchor(event ? (event.target || event.srcElement) : null);
    if (!anchor || !anchor.getAttribute) {
        return;
    }

    // Важно читать именно атрибут href после выполнения inline onclick:
    // функции prepare* динамически заменяют javascript:void(0) на panelkasettings://... .
    var rawHref = anchor.getAttribute("data-href") || anchor.getAttribute("href") || "";
    if (rawHref.indexOf("panelkasettings://") !== 0) {
        return;
    }

    if (event.preventDefault) {
        event.preventDefault();
    }
    event.returnValue = false;

    if (rawHref === "panelkasettings://open_additional_settings" && additionalOpen) {
        closeAdditionalSettings();
        return;
    }
    if (tryLocalSource(rawHref) || tryLocalSelection(rawHref)) return;
    rawHref = appendScrollState(rawHref);
    publishBridgeCommand(rawHref);
}

if (document.addEventListener) {
    document.addEventListener("click", handleBridgeClick, false);
} else if (document.attachEvent) {
    document.attachEvent("onclick", handleBridgeClick);
}

function findPageByName(name) {
    var i;
    for (i = 0; i < DATA.Pages.length; i++) {
        if (DATA.Pages[i].Name === name) { return DATA.Pages[i]; }
    }
    return null;
}

function findColumnByName(name, pageName) {
    var i, j, page, column;
    // ID is authoritative; a name is resolved only within the displayed page.
    for (i = 0; i < DATA.Pages.length; i++) {
        page = DATA.Pages[i];
        for (j = 0; j < page.Columns.length; j++) {
            column = page.Columns[j];
            if (column.Id === name) return column;
        }
    }
    pageName = pageName || DATA.Ui.ActivePage;
    for (i = 0; i < DATA.Pages.length; i++) {
        page = DATA.Pages[i];
        if (page.Name !== pageName) continue;
        for (j = 0; j < page.Columns.length; j++) {
            if (page.Columns[j].Name === name) return page.Columns[j];
        }
    }
    return null;
}

var contextMenuReturnFocus = null;
function openContextMenu(event, kind, pageName, columnName, source, index, groupName) {
    if (bridgeBusy) return false;
    event = event || window.event;
    if (event.preventDefault) { event.preventDefault(); }
    event.returnValue = false;
    if (event.stopPropagation) { event.stopPropagation(); }
    event.cancelBubble = true;

    var menu = document.getElementById("panelkaContextMenu");
    var html = "";
    var page, column, item;

    if (kind === "page") {
        page = findPageByName(pageName);
        if (page && !boolValue(page.System)) {
            html += contextItem(uiText("moveEarlier"), actionUrl("move_page", { page: pageName, direction: "prev" }));
            html += contextItem(uiText("moveLater"), actionUrl("move_page", { page: pageName, direction: "next" }));
            html += '<div class="context-separator"></div>';
            html += '<a role="menuitem" class="context-item danger" href="javascript:void(0)" onclick="closeContextMenu(); return requestPageDelete(\'' + escapeJs(pageName) + ("');\">" + escapeHtml(uiText("deletePageMenu")) + "</a>");
        }
    } else if (kind === "column") {
        column = findColumnByName(columnName);
        if (column && !boolValue(column.System)) {
            html += contextItem(uiText("moveHigher"), actionUrl("move_column", { column: columnName, direction: "prev" }));
            html += contextItem(uiText("moveLower"), actionUrl("move_column", { column: columnName, direction: "next" }));
            html += '<div class="context-separator"></div>';
            html += '<a role="menuitem" class="context-item danger" href="javascript:void(0)" onclick="closeContextMenu(); return requestColumnDelete(\'' + escapeJs(columnName) + ("');\">" + escapeHtml(uiText("deleteColumnMenu")) + "</a>");
        }
    } else if (kind === "group") {
        column = findColumnByName(columnName);
        if (column && boolValue(DATA.General.GroupByType)) {
            html += contextItem(uiText("moveGroupHigher"), actionUrl("move_group", {columnId:column.Id, group:groupName, direction:"prev"}));
            html += contextItem(uiText("moveGroupLower"), actionUrl("move_group", {columnId:column.Id, group:groupName, direction:"next"}));
        }
    } else if (kind === "link") {
        html += contextItem(uiText("moveHigher"), actionUrl("move_link", { source: source, index: index, column: columnName, direction: "prev" }));
        html += contextItem(uiText("moveLower"), actionUrl("move_link", { source: source, index: index, column: columnName, direction: "next" }));
        html += '<div class="context-separator"></div>';
        if (columnName && columnName !== "Не распределено") {
            html += contextItem(uiText("removeFromColumn"), actionUrl("remove_link_from_column", { source: source, index: index, column: columnName }));
        }
        html += contextItem(uiText("moveToUnassigned"), actionUrl("unassign_link", {source: source, index: index, column: columnName}));
        item = findLinkData(source, index);
        html += '<a role="menuitem" class="context-item danger" href="javascript:void(0)" onclick="return requestLinkDelete(\'' + escapeJs(source) + '\', ' + index + (");\">" + escapeHtml(uiText("deleteLinkMenu")) + "</a>");
    }

    if (!html) { return false; }
    contextMenuReturnFocus = document.activeElement;
    menu.innerHTML = html;
    menu.style.display = "block";
    var target = event.target || event.srcElement;
    var anchorRect = event.keyboard && target && target.getBoundingClientRect ? target.getBoundingClientRect() : null;
    var x = anchorRect ? anchorRect.left : (event.clientX || 0);
    var y = anchorRect ? anchorRect.bottom : (event.clientY || 0);
    var first = menu.querySelector ? menu.querySelector('a') : null;
    if (first && first.focus) first.focus();
    menu.style.left = x + "px";
    menu.style.top = y + "px";
    window.setTimeout(function () {
        var rect = menu.getBoundingClientRect ? menu.getBoundingClientRect() : null;
        if (rect) {
            if (rect.right > document.documentElement.clientWidth) { menu.style.left = Math.max(4, x - rect.width) + "px"; }
            if (rect.bottom > document.documentElement.clientHeight) { menu.style.top = Math.max(4, y - rect.height) + "px"; }
        }
    }, 0);
    return false;
}

// Drag payloads stay inside this form; external text/URL drops never execute commands.
var settingsDrag = null, settingsDropNode = null;
function dragDescriptor(node, drop) {
    while (node && node !== document.body) {
        if (node.getAttribute) {
            var kind = node.getAttribute(drop ? 'data-drop-kind' : 'data-drag-kind');
            var id = node.getAttribute(drop ? 'data-drop-id' : 'data-drag-id');
            if (!kind && drop) {kind=node.getAttribute('data-drag-kind');id=node.getAttribute('data-drag-id');}
            if (kind) return {kind:kind,id:id,group:node.getAttribute('data-group-name')||'',node:node};
        }
        node = node.parentNode;
    }
    return null;
}
function clearSettingsDrop() {
    if(settingsDropNode) settingsDropNode.className=settingsDropNode.className.replace(/ panelka-drop-(before|inside)/g,'');
    settingsDropNode=null;
}
function validSettingsDrop(from,to) {
    if(!from||!to)return false;
    if(from.kind===to.kind && from.id===to.id && from.group===to.group)return false;
    if(from.kind==='Group')return from.id===to.id && (to.kind==='Group'||to.kind==='Column');
    if(from.kind==='Link')return (to.kind==='Link'||to.kind==='Column') && to.id!=='column-standard' && to.id!=='column-unassigned';
    if(from.kind==='Column')return (to.kind==='Column'||to.kind==='Page') && to.id!=='page-standard' && to.id!=='page-unassigned' && to.id!=='column-standard' && to.id!=='column-unassigned';
    return false;
}
function startSettingsDrag(event) {
    event=event||window.event;
    var from=dragDescriptor(event.target||event.srcElement,false);
    if(bridgeBusy || !from || from.node.getAttribute('draggable')!=='true'){if(event.preventDefault)event.preventDefault();return;}
    closeContextMenu();settingsDrag=from;document.body.className=(document.body.className||'')+' panelka-dragging';
    if(event.dataTransfer){event.dataTransfer.effectAllowed='move';event.dataTransfer.setData('Text','panelka-settings');}
    if(event.stopPropagation)event.stopPropagation();
}
function overSettingsDrag(event) {
    event=event||window.event;
    var to=dragDescriptor(event.target||event.srcElement,true);
    clearSettingsDrop();
    if(!validSettingsDrop(settingsDrag,to))return;
    if(event.preventDefault)event.preventDefault();event.returnValue=false;
    if(event.dataTransfer)event.dataTransfer.dropEffect='move';
    var shell=document.getElementById('previewShell');
    if(shell && shell.getBoundingClientRect){
        var rect=shell.getBoundingClientRect();
        if(event.clientY<rect.top+35)shell.scrollTop-=18;
        else if(event.clientY>rect.bottom-35)shell.scrollTop+=18;
    }
    settingsDropNode=to.node;
    settingsDropNode.className+=(to.kind===settingsDrag.kind?' panelka-drop-before':' panelka-drop-inside');
}
function dropSettingsDrag(event) {
    event=event||window.event;
    var from=settingsDrag,to=dragDescriptor(event.target||event.srcElement,true);
    clearSettingsDrop();settingsDrag=null;document.body.className=(document.body.className||'').replace(/ panelka-dragging/g,'');
    if(!validSettingsDrop(from,to))return;
    if(event.preventDefault)event.preventDefault();event.returnValue=false;
    if(event.stopPropagation)event.stopPropagation();
    publishBridgeCommand(appendScrollState(actionUrl('drag_structure',{
        kind:from.kind,dragId:from.id,group:from.group,targetKind:to.kind,targetId:to.id,targetGroup:to.group
    })));
}
if(document.addEventListener){
    document.addEventListener('dragstart',startSettingsDrag,false);
    document.addEventListener('dragover',overSettingsDrag,false);
    document.addEventListener('drop',dropSettingsDrag,false);
    document.addEventListener('dragend',function(){settingsDrag=null;clearSettingsDrop();document.body.className=(document.body.className||'').replace(/ panelka-dragging/g,'');},false);
}

function findLinkData(source, index) {
    var i, j, k, links, item;
    for (i = 0; i < DATA.Pages.length; i++) {
        for (j = 0; j < DATA.Pages[i].Columns.length; j++) {
            links = DATA.Pages[i].Columns[j].Links || [];
            for (k = 0; k < links.length; k++) {
                item = links[k];
                if (item.Source === source && parseInt(item.Index, 10) === parseInt(index, 10)) { return item; }
            }
        }
    }
    return null;
}

function findContextTarget(node) {
    while (node && node !== document.body) {
        if (node.getAttribute && node.getAttribute("data-context-kind")) { return node; }
        node = node.parentNode;
    }
    return null;
}

function handleContextMenuEvent(event) {
    event = event || window.event;
    var target = findContextTarget(event.target || event.srcElement);
    if (event.preventDefault) { event.preventDefault(); }
    event.returnValue = false;
    if (!target) {
        closeContextMenu();
        return false;
    }
    return openContextMenu(
        event,
        target.getAttribute("data-context-kind") || "",
        target.getAttribute("data-page-name") || "",
        target.getAttribute("data-column-name") || "",
        target.getAttribute("data-source") || "",
        parseInt(target.getAttribute("data-index") || "-1", 10), target.getAttribute("data-group-name") || "");
}

    document.attachEvent("onclick", function () { closeContextMenu(); });

}

// The server classifies links and filters metadata groups. The client only hides
// unavailable entry points; every addition is checked again on the server.
function personalSourceAllowed(key) {
    if (!boolValue(DATA.Ui.Personal) && key !== "ЛокальныеФайлы" && key !== "СтандартныеИнструменты") return true;
    var policy = DATA.Ui.AllowedSources;
    return !policy || !policy.hasOwnProperty(key) || boolValue(policy[key]);
}

function hasPersonalSourceRestrictions() {
    if (!boolValue(DATA.Ui.Personal)) return false;
    var policy = DATA.Ui.AllowedSources || {}, key;
    for (key in policy) if (policy.hasOwnProperty(key) && !boolValue(policy[key])) return true;
    return false;
}

function metadataSourcesAllowed() {
    // Older servers omit the policy and retain their existing behaviour.
    return !boolValue(DATA.Ui.Personal) || !DATA.Ui.AllowedSources || DATA.MetadataGroups.length > 0;
}

function metadataGroupAvailable(group) {
    if (!boolValue(DATA.Ui.Personal) || !DATA.Ui.AllowedSources) return true;
    for (var i = 0; i < DATA.MetadataGroups.length; i++) if (DATA.MetadataGroups[i].Id === group) return true;
    return false;
}

function extraSourcesAllowed() {
    return personalSourceAllowed("ДополнительныеОбработки") && boolValue(DATA.Ui.HasExtraProcessing);
}
function manualLinkAllowed() {
    return personalSourceAllowed("ВебСсылки") || metadataSourcesAllowed() || extraSourcesAllowed();
}
function moreSourcesAllowed() {
    return metadataSourcesAllowed() || personalSourceAllowed("ВебСсылки") || personalSourceAllowed("ЛокальныеФайлы") ||
        personalSourceAllowed("СтандартныеИнструменты") ||
        (personalSourceAllowed("ДополнительныеОбработки") && boolValue(DATA.Ui.HasExtraProcessing));
}

function refreshMetadataCachePolicy() {
    var groups = DATA.MetadataGroups || [], ids = [], i;
    for (i = 0; i < groups.length; i++) ids.push(groups[i].Id);
    var signature = JSON.stringify([boolValue(DATA.Ui.Personal), DATA.Ui.AllowedSources || null, ids]);
    if (signature !== metadataSourceCachePolicy) {
        metadataSourceCache = {};
        metadataSourceCachePolicy = signature;
    }
}

function applyLinkResponse(delta) {
    var i,j,k,item,changed={},updated={},dirtyColumns={},catalog=DATA.Catalog||[],selected=DATA.Selected;
    for(i=0;i<catalog.length;i++) if(catalog[i].Id===delta.Id) {
        catalog[i].Name=delta.Name; catalog[i].Tooltip=delta.Tooltip;
    }
    for(i=0;i<delta.Placements.length;i++) changed['id:'+delta.Placements[i].PlacementId]=delta.Placements[i];
    for(i=0;i<DATA.Pages.length;i++) for(j=0;j<DATA.Pages[i].Columns.length;j++) {
        var column=DATA.Pages[i].Columns[j], links=column.Links;
        for(k=0;k<links.length;k++) {
            item=links[k]; var key='id:'+item.PlacementId, value=changed[key];
            if(!value || (item.Name===value.Name && item.Tooltip===value.Tooltip && item.Custom===value.Custom &&
                item.CustomName===value.CustomName && item.CustomTooltip===value.CustomTooltip))continue;
            dirtyColumns['column:'+column.Id]=true;
            updated[key]=value;
            item.Name=value.Name;item.Tooltip=value.Tooltip;item.Custom=value.Custom;
            item.CustomName=value.CustomName;item.CustomTooltip=value.CustomTooltip;
        }
    }
    var rows=document.querySelectorAll('.link-row');
    for(i=0;i<rows.length;i++) {
        item=updated['id:'+rows[i].getAttribute('data-placement-id')];
        if(item)rows[i].innerHTML=renderLinkContent(item);
    }
    if(selected.Kind==='Link' && selected.Id===delta.Id) {
        selected.Name=delta.Name;selected.Tooltip=delta.Tooltip;
        item=changed['id:'+selected.Placement.Id];
        if(item) {
            selected.Placement.Name=item.Name;selected.Placement.Tooltip=item.Tooltip;
            selected.Placement.CustomName=item.CustomName;selected.Placement.CustomTooltip=item.CustomTooltip;
        }
        renderProperties();editorBaseline=JSON.stringify(currentEditorPayload());updateEditorDraft();
    }
    if (DATA.Ui.SourceMode === "catalog") {
        for (i=0;i<DATA.Sources.length;i++) if (DATA.Sources[i].Id===delta.Id) {
            DATA.Sources[i].Name=delta.Name;DATA.Sources[i].Tooltip=delta.Tooltip;
        }
        renderSourceList();
    }
    DATA.Ui.State=delta.State;restoreViewState();
    refreshBoardMarkupCache(dirtyColumns);
}

function renderLinks(links, columnName, columnId) {
    var html = "";
    var i, j, item, lastType = null;
    var ordered = (links || []).slice(0);

    ordered.sort(function (a, b) {
        var ao = parseInt(valueOr(a.Order, 1000000), 10);
        var bo = parseInt(valueOr(b.Order, 1000000), 10);
        return ao - bo;
    });

    // При включенной группировке каждый тип всегда отображается единым блоком.
    // Порядок самих групп определяется первым появлением типа в ручном порядке,
    // а порядок ссылок внутри группы сохраняет пользовательскую сортировку.
    if (boolValue(DATA.General.GroupByType) && ordered.length > 1) {
        var typeOrder = [];
        var grouped = {};
        var groupedOrdered = [];
        var typeKey;

        for (i = 0; i < ordered.length; i++) {
            typeKey = String(valueOr(ordered[i].Type, ""));
            if (!grouped.hasOwnProperty(typeKey)) {
                grouped[typeKey] = [];
                typeOrder.push(typeKey);
            }
            grouped[typeKey].push(ordered[i]);
        }

        for (i = 0; i < typeOrder.length; i++) {
            typeKey = typeOrder[i];
            for (j = 0; j < grouped[typeKey].length; j++) {
                groupedOrdered.push(grouped[typeKey][j]);
            }
        }

        ordered = groupedOrdered;
    }

    if (ordered.length === 0) {
        return ("<div class=\"empty-column\">" + escapeHtml(uiText("emptyColumn")) + "</div>");
    }

    for (i = 0; i < ordered.length; i++) {
        item = ordered[i];
        if (boolValue(DATA.General.GroupByType) && item.Type !== lastType) {
            lastType = item.Type;
            html += '<div class="type-head" tabindex="0" role="button" aria-haspopup="menu" aria-label="' + escapeHtml(uiText('groupActions',{group:lastType})) + '" draggable="true" data-drag-kind="Group" data-drag-id="' + escapeHtml(columnId) + '" data-context-kind="group" data-column-name="' + escapeHtml(columnName) + '" data-group-name="' + escapeHtml(lastType) + ("\" title=\"" + escapeHtml(uiText("groupDragHint")) + "\">") + escapeHtml(lastType) + '</div>';
        }
        html += '<a class="link-row' + (item.Selected ? ' selected' : '') + '" href="' + actionUrl("select_link", { source: item.Source, index: item.Index, column: columnName, id:item.Id, placementId:item.PlacementId }) + '"' +
            ' draggable="true" data-drag-kind="Link" data-drag-id="' + escapeHtml(item.PlacementId) + '" data-placement-id="' + escapeHtml(item.PlacementId) + '" data-context-kind="link" data-column-name="' + escapeHtml(columnName) + '" data-source="' + escapeHtml(item.Source) + '" data-index="' + item.Index + '">' +
            renderLinkContent(item) + '</a>';
    }
    return html;
}

// Small local patches already changed these nodes; remember their current model representation.
function refreshBoardMarkupCache(dirtyColumns) {
    var page = getActivePage(), i, key, cached;
    if (!page) return;
    for (i = 0; i < page.Columns.length; i++) {
        key = 'column:' + page.Columns[i].Id;
        cached = boardColumnNodes[key];
        if (dirtyColumns[key] && cached) cached.markup = columnMarkup(page.Columns[i], page);
    }
}

function renderBoard(columnsToUpdate) {
    var host = document.getElementById("board"), page = getActivePage();
    var next = {}, i, key, cached, markup, wrapper, node;
    if (!page) {
        document.getElementById("activePageCrumb").innerHTML = escapeHtml(uiText("noPages"));
        document.getElementById("columnsCrumb").innerHTML = escapeHtml(uiText("zeroColumns"));
        host.innerHTML = ("<div class=\"empty-board\">" + escapeHtml(uiText("firstPageHelp")) + "</div>");
        boardColumnNodes = {};
        return;
    }
    document.getElementById("activePageCrumb").innerHTML = escapeHtml(page.Name);
    document.getElementById("columnsCrumb").innerHTML = escapeHtml(uiText('columnCount',{count:page.Columns.length}));
    for (i = 0; i <= page.Columns.length; i++) {
        if (i < page.Columns.length) {
            key = 'column:' + page.Columns[i].Id;
            cached = boardColumnNodes[key];
            markup = columnsToUpdate && columnsToUpdate.indexOf(page.Columns[i].Id) < 0 && cached && cached.node.parentNode === host ?
                cached.markup : columnMarkup(page.Columns[i], page);
        } else {
            key = 'footer';
            markup = page.Name !== "Стандартные инструменты" && page.Name !== "Не распределено" ?
                '<a class="new-column" href="' + actionUrl("add_column_default", { page: page.Name }) + '">' + escapeHtml(uiText('addColumnToPage',{page:page.Name})) + '</a>' :
                ("<div class=\"system-column-note\">" + escapeHtml(uiText("systemColumnsHelp")) + "</div>");
        }
        cached = boardColumnNodes[key];
        if (cached && cached.markup === markup && cached.node.parentNode === host) {
            node = cached.node;
        } else {
            wrapper = document.createElement('div');
            wrapper.innerHTML = markup;
            node = wrapper.firstChild;
            if (cached && cached.node.parentNode === host) host.removeChild(cached.node);
        }
        if (host.children[i] !== node) host.insertBefore(node, host.children[i] || null);
        next[key] = {node:node, markup:markup};
    }
    while (host.children.length > page.Columns.length + 1) host.removeChild(host.lastChild);
    boardColumnNodes = next;
}

function commandParameters(url) {
    var params={}, parts=(url.split('?')[1]||'').split('&');
    for(var i=0;i<parts.length;i++) {
        var at=parts[i].indexOf('=');if(at<0)continue;
        var value=parts[i].slice(at+1).replace(/-/g,'+').replace(/_/g,'/').replace(/\./g,'=');
        params[parts[i].slice(0,at)]=decodeURIComponent(escape(window.atob(value)));
    }
    return params;
}

function additionalButtonIcon(action) {
    var icon = additionalButtonIcons[action];
    return icon ? '<img class="additional-icon" alt="" src="data:image/png;base64,' + icon + '">' : '';
}

function additionalButton(label, action) {
    return '<a class="btn" href="' + actionUrl(action) + '">' + additionalButtonIcon(action) +
        escapeHtml(uiText(label)) + '</a>';
}

function renderAdditionalTabs() {
    var host = document.getElementById('additionalTabs');
    if (!host) return;
    var html = '', admin = DATA.Additional && DATA.Additional.CanAdmin;
    for (var i = 0; i < additionalTabs.length; i++) {
        var tab = additionalTabs[i];
        if (tab.id === 'personal' && !admin) continue;
        var active = additionalTab === tab.id;
        html += '<button type="button" class="additional-tab" id="additional-tab-' + tab.id + '" role="tab" aria-controls="additionalBody" ' +
            'aria-selected="' + active + '" tabindex="' + (active ? '0' : '-1') + '" onclick="selectAdditionalTab(\'' + tab.id + '\', true)" ' +
            'onkeydown="return additionalTabKey(event, \'' + tab.id + '\')"><img class="additional-icon" alt="" src="data:image/png;base64,' +
            tab.icon + '">' + escapeHtml(uiText(tab.title)) + '</button>';
    }
    host.innerHTML = html;
}

function additionalTabKey(event, id) {
    var key = event.keyCode;
    if (key !== 37 && key !== 39 && key !== 36 && key !== 35) return true;
    var ids = ['display','appearance','transfer'];
    if (DATA.Additional && DATA.Additional.CanAdmin) ids.push('personal');
    var index = ids.indexOf(id);
    index = key === 36 ? 0 : key === 35 ? ids.length - 1 : (index + ids.length + (key === 39 ? 1 : -1)) % ids.length;
    if (event.preventDefault) event.preventDefault();
    selectAdditionalTab(ids[index], true);
    return false;
}

function selectAdditionalTab(id, focus) {
    if (bridgeBusy) return;
    additionalTab = id;
    renderAdditionalTabs();
    renderAdditionalBody();
    updateViewStateField();
    var tab = focus && document.getElementById('additional-tab-' + id);
    if (tab && tab.focus) tab.focus();
    if (id === 'personal' && DATA.Additional.CanManageProfiles && !DATA.Additional.hasOwnProperty('Profiles')) {
        publishBridgeCommand('panelkasettings://additional_profiles');
    }
}

function additionalField(index, field) {
    var id = 'additional-field-' + index, label = escapeHtml(field.Label);
    if (field.Kind === 'bool') {
        return '<label class="additional-check"><input type="checkbox" id="' + id + '"' + (field.Value ? ' checked' : '') +
            ' onchange="changeAdditionalField(' + index + ',this.checked)">' + label + '</label>';
    }
    if (field.Kind === 'tooltip') {
        var values = ['Снизу','Всплывающая','Кнопка','Нет'];
        var titles = ['additionalTooltipBottom','additionalTooltipPopup','additionalTooltipButton','additionalTooltipNone'];
        var html = '<label for="' + id + '">' + label + '</label> <select id="' + id + '" onchange="changeAdditionalField(' + index + ',this.value)">';
        for (var i = 0; i < values.length; i++) {
            html += '<option value="' + values[i] + '"' + (values[i] === field.Value ? ' selected' : '') + '>' + escapeHtml(uiText(titles[i])) + '</option>';
        }
        return html + '</select>';
    }
    return '';
}

function additionalFieldGroup(group, columnCount) {
    var fields = DATA.Additional.Fields || [], columns = [], html = '';
    for (var i = 0; i < columnCount; i++) columns.push([]);
    for (var j = 0; j < fields.length; j++) {
        var field = fields[j];
        if (field.Group !== group) continue;
        var column = Math.min(columnCount, Math.max(1, Number(field.Column) || 1));
        columns[column - 1].push(j);
    }
    for (var k = 0; k < columns.length; k++) {
        var indexes = columns[k];
        indexes.sort(function(a,b) { return (fields[a].Order || 0) - (fields[b].Order || 0) || a - b; });
        html += '<div class="additional-field-column">';
        for (var n = 0; n < indexes.length; n++) html += additionalField(indexes[n], fields[indexes[n]]);
        html += '</div>';
    }
    return html;
}

function renderAdditionalDisplayGroup(section, title) {
    var fields = DATA.Additional.Fields || [], indexes = [];
    for (var i = 0; i < fields.length; i++) {
        if (fields[i].Group === 'display' && fields[i].Section === section) indexes.push(i);
    }
    indexes.sort(function(a,b) { return (fields[a].Order || 0) - (fields[b].Order || 0); });
    var html = '<section class="additional-card additional-stack"><h3>' + escapeHtml(uiText(title)) + '</h3>';
    for (var j = 0; j < indexes.length; j++) html += additionalField(indexes[j],fields[indexes[j]]);
    return html + '</section>';
}

function renderAdditionalAppearanceField(field) {
    var isColor = field.Kind === 'color';
    var label = uiText(isColor ? 'additionalColor' : 'additionalFont');
    var value = field.Presentation || uiText('additionalDefault');
    var color = isColor && /^#[0-9a-f]{6}$/i.test(field.Value || '') ? field.Value : '';
    return '<div class="additional-appearance-field"><span>' + escapeHtml(label) + '</span>' +
        '<a class="btn additional-value" href="' + actionUrl('additional_choose_appearance',{key:field.Key}) +
        '" aria-label="' + escapeHtml(field.Label + ': ' + value) + '" title="' + escapeHtml(value) + '">' +
        (color ? '<span class="additional-color-swatch" aria-hidden="true" style="background-color:' + color + '"></span>' : '') +
        '<span class="additional-value-text">' + escapeHtml(value) + '</span><span class="additional-ellipsis" aria-hidden="true">…</span></a></div>';
}

function renderAdditionalAppearance() {
    var fields = DATA.Additional.Fields || [], sections = ['column','group'], html = '';
    for (var i = 0; i < sections.length; i++) {
        var section = sections[i], font = null, color = null;
        for (var j = 0; j < fields.length; j++) {
            var field = fields[j];
            if (field.Group !== 'appearance' || field.Section !== section) continue;
            if (field.Kind === 'font') font = field;
            else if (field.Kind === 'color') color = field;
        }
        var title = section === 'column' ? 'additionalColumnHeadings' : 'additionalGroupHeadings';
        html += '<section class="additional-card"><h3>' + escapeHtml(uiText(title)) + '</h3><div class="additional-appearance-fields">' +
            (font ? renderAdditionalAppearanceField(font) : '') + (color ? renderAdditionalAppearanceField(color) : '') + '</div></section>';
    }
    return html;
}

// Keep checkbox edits in the current document until a real 1C action needs them.
function readAdditionalDraft() {
    var field = document.getElementById('panelkaAdditionalDraft');
    return field && field.value ? JSON.parse(field.value) : {};
}

function changeAdditionalField(index, value) {
    var field = DATA.Additional.Fields[index];
    var draftField = document.getElementById('panelkaAdditionalDraft');
    if (!field || !draftField || bridgeBusy || field.Value === value) return;
    var draft = readAdditionalDraft();
    draft[field.Key] = value;
    draftField.value = JSON.stringify(draft);
    field.Value = value;
    additionalNeedsRefresh = true;
}

function setAdditionalCategories(value) {
    if (bridgeBusy) return;
    var fields = DATA.Additional.Fields || [];
    for (var i = 0; i < fields.length; i++) {
        if (fields[i].Group !== 'categories' || fields[i].Kind !== 'bool') continue;
        changeAdditionalField(i, value);
        var input = document.getElementById('additional-field-' + i);
        if (input) input.checked = value;
    }
}

function additionalProfileCommands() {
    var profiles = DATA.Additional.Profiles || [], selected;
    for (var i = 0; i < profiles.length; i++) if (profiles[i].User === additionalProfile) selected = profiles[i];

    if (!host) return;
    var disabled = selected ? '' : ' disabled';
    host.innerHTML = additionalButton('additionalRefresh','additional_profiles') +

        additionalButtonIcon('additional_preview') + escapeHtml(uiText('additionalPreview')) + '</button>' +
        '<button class="btn" type="button" onclick="runAdditionalProfile(\'additional_export_profile\')"' + disabled + '>' +
        additionalButtonIcon('additional_export_profile') + escapeHtml(uiText('additionalExport')) + '</button>' + (DATA.Additional.CanDeleteProfiles ?
        '<button class="btn btn-danger" type="button" onclick="runAdditionalProfile(\'additional_delete_profile\')"' + disabled + '>' +

}

function runAdditionalProfile(action) {
    var profiles = DATA.Additional.Profiles || [];
    for (var i = 0; i < profiles.length; i++) {
        var row = profiles[i];
        if (row.User === additionalProfile) {
            publishBridgeCommand(actionUrl(action,{user:row.User,revision:row.Revision}));
            return;
        }
    }
}

function filterAdditionalProfiles(value) {
    additionalQuery = value;
    var table = document.getElementById('additionalProfilesTable');
    if (!table) return;
    var rows = table.querySelectorAll('tbody tr');
    var query = value.toLowerCase();
    for (var i = 0; i < rows.length; i++) {
        var visible = (rows[i].getAttribute('data-profile-name') || '').toLowerCase().indexOf(query) >= 0;
        rows[i].style.display = visible ? '' : 'none';
        var radio = rows[i].querySelector('input');
        if (!visible && radio && radio.checked) {
            radio.checked = false;
            additionalProfile = '';
            additionalProfileCommands();
        }
    }
}

    if (!pane || !workspace || !toggle) return;

function restoreAdditionalView() {

}

function receiveAdditionalResponse() {
    var mailbox = document.getElementById('panelkaAdditionalResponse');
    if (!mailbox || !mailbox.value) return false;
    var raw = mailbox.value, snapshot;
    mailbox.value = '';
    try { snapshot = JSON.parse(raw); } catch (e) { showToast(uiText('updateFailed')); return false; }
    if (!snapshot || typeof snapshot !== 'object') return false;
    DATA.Additional = DATA.Additional || {};
    for (var name in snapshot) {
        if (Object.prototype.hasOwnProperty.call(snapshot,name)) DATA.Additional[name] = snapshot[name];
    }
    if (snapshot.Fields) {
        var draft = readAdditionalDraft();
        for (var i = 0; i < snapshot.Fields.length; i++) {
            var field = snapshot.Fields[i];
            if (Object.prototype.hasOwnProperty.call(draft,field.Key)) field.Value = draft[field.Key];
        }
    }
    if (snapshot.General && JSON.stringify(DATA.General) !== JSON.stringify(snapshot.General)) {
        DATA.General = snapshot.General;
        additionalNeedsRefresh = true;
    }
    if (activeBridgeAction === 'open_additional_settings' && additionalOpeningDraft) {
        // The server has already applied the workspace draft while opening the tabs.
        editorBaseline = JSON.stringify(currentEditorPayload());
        updateEditorDraft();
        additionalNeedsRefresh = true;
        additionalOpeningDraft = false;
    }
    if (!additionalOpen && activeBridgeAction === 'open_additional_settings') showAdditionalSettings(true);
    else if (additionalOpen) {
        renderAdditionalTabs();
        renderAdditionalBody();
    }
    return true;
}

function currentEditorPayload() {
    var selected = DATA.Selected || {}, payload;
    if (selected.Kind === 'Link') {
        payload = linkEditorPayload();
        if (payload) payload.kind = 'Link';
        return payload;
    }
    if (selected.System) return null;
    if (selected.Kind === 'Page' && document.getElementById('pageName')) {
        return {kind:'Page', id:selected.Id, name:document.getElementById('pageName').value};

    if (selected.Kind === 'Column' && document.getElementById('columnName')) {
        return {kind:'Column', id:selected.Id, name:document.getElementById('columnName').value, page:document.getElementById('columnPage').value};
    }
    return null;
}

// The server writes JSON into a hidden input; no eval or document navigation is used.
function receiveServerResponse() {
    var mailbox = document.getElementById('panelkaServerResponse');
    if (!mailbox || !mailbox.value) return false;
    var response;
    try {
        response = JSON.parse(mailbox.value);
        if (!response || !response.Data || (response.Kind !== 'sources' && response.Kind !== 'model' && response.Kind !== 'layout' && response.Kind !== 'patch' && response.Kind !== 'link')) {
            throw new Error('Unknown response');
        }
    } catch (error) {
        mailbox.value = '';
        showToast(uiText("updateFailed"));
        return false;
    }
    mailbox.value = '';
    // The bridge has one command in flight. Revision additionally rejects an older model reply.
    if (typeof response.Revision === 'number' && response.Revision < lastModelRevision) return true;
    if (typeof response.Revision === 'number') lastModelRevision = response.Revision;
    var webLinkSubmitted = completeWebLinkSubmission();
    var renderingStarted=clockMs();
    if (response.Kind === 'link') {
        applyLinkResponse(response.Data);
        if (webLinkSubmitted) renderSourceList();
    } else if (response.Kind === 'sources') {
        DATA.Sources = response.Data.Sources;
        if (response.Data.hasOwnProperty('AllowedSources')) DATA.Ui.AllowedSources = response.Data.AllowedSources;
        if (response.Data.hasOwnProperty('MetadataGroups')) DATA.MetadataGroups = response.Data.MetadataGroups;
        DATA.Ui.SourceMode = response.Data.SourceMode;
        DATA.Ui.MetadataGroup = response.Data.MetadataGroup;
        DATA.Ui.FavoriteFilter = response.Data.FavoriteFilter;
        DATA.Ui.State = response.Data.State;
        rememberMetadataSources();
        renderSourceTabs();
        renderMetadataGroups();
        renderSourceList();
        restoreViewState();
    } else {
        var previousUi = {page:DATA.Ui.ActivePage,column:DATA.Ui.ActiveColumn,source:DATA.Ui.SourceMode};
        if (response.Kind === 'patch') {
            applyStructurePatch(response.Data);
        } else if (response.Kind === 'layout') {
            for (var name in response.Data) if (response.Data.hasOwnProperty(name)) DATA[name] = response.Data[name];
        } else {
            DATA = response.Data;
        }
        renderServerChanges(response.Changes, previousUi, webLinkSubmitted);
    }
    if(pendingPerformance) {
        recordPerformance({action:pendingPerformance.action,local:false,kind:response.Kind,
            totalMs:clockMs()-pendingPerformance.started,renderMs:clockMs()-renderingStarted,
            observedQueueMs:pendingPerformance.processing ? pendingPerformance.processing-pendingPerformance.started : null,
            serverMs:typeof response.ServerMs==='number' ? response.ServerMs : null});
        pendingPerformance=null;
    }
    return true;
}

// Updated columns carry Links; unchanged column headers deliberately omit this property.
// Catalog deletions use the complete layout response to keep legacy numeric indexes consistent.
function applyStructurePatch(data) {
    var oldPages = {}, oldColumns = {}, catalog = {}, i, j, page, column, old, key;
    for (i = 0; i < DATA.Pages.length; i++) {
        page = DATA.Pages[i];
        oldPages['id:' + page.Id] = page;
        for (j = 0; j < page.Columns.length; j++) {
            column = page.Columns[j];
            oldColumns['id:' + column.Id] = column;
        }
    }
    for (i = 0; i < data.Pages.length; i++) {
        page = data.Pages[i];
        old = oldPages['id:' + page.Id];
        if (!page.hasOwnProperty('IconData') && old && old.Icon === page.Icon) page.IconData = old.IconData;
        for (j = 0; j < page.Columns.length; j++) {
            column = page.Columns[j];
            old = oldColumns['id:' + column.Id];
            if (!column.hasOwnProperty('IconData') && old && old.Icon === column.Icon) column.IconData = old.IconData;
            if (!column.hasOwnProperty('Links')) {
                if (!old) throw new Error('Missing column in structure patch');
                column.Links = old.Links;
            }
        }
    }
    var mergedCatalog = (DATA.Catalog || []).slice(0);
    for (i = 0; i < mergedCatalog.length; i++) catalog['id:' + mergedCatalog[i].Id] = i;
    for (i = 0; i < (data.Catalog || []).length; i++) {
        key = 'id:' + data.Catalog[i].Id;
        if (typeof catalog[key] === 'number') mergedCatalog[catalog[key]] = data.Catalog[i];
        else {catalog[key] = mergedCatalog.length;mergedCatalog.push(data.Catalog[i]);}
    }
    for (key in data) if (data.hasOwnProperty(key) && key !== 'Catalog') DATA[key] = data[key];
    DATA.Catalog = mergedCatalog;
}

// Explicit server contract replaces serialization/comparison of entire visible regions.
function renderServerChanges(changes, previousUi, webLinkSubmitted) {
    if (!changes) {renderAll();return;} // Compatibility with responses created before the contract.
    normalizeData();
    if (changes.Sources) {
        metadataSourceCache = {};
        rememberMetadataSources();
        renderSourceTabs();renderMetadataGroups();renderSourceList();
    } else if (webLinkSubmitted || previousUi.column !== DATA.Ui.ActiveColumn || previousUi.source !== DATA.Ui.SourceMode) {
        renderSourceList();
    }
    if (changes.Structure) {
        if (changes.Headers || previousUi.page !== DATA.Ui.ActivePage) renderPageTabs();
        var columns = !changes.Full && previousUi.page === DATA.Ui.ActivePage && previousUi.column === DATA.Ui.ActiveColumn ? changes.Columns : null;
        renderBoard(columns);
    }
    if (changes.Properties) renderProperties();
    restoreScrollState();
    editorBaseline = JSON.stringify(currentEditorPayload());
    updateEditorDraft();restoreViewState();
}

}
