        /** Формирование URL-safe Base64 строки для передачи команд в 1С */
        // Названия и подсказки — обычный текст, как в интерфейсе 1С.
        window.escapePbdText = function(text) {
            return String(text || '').replace(/&/g, '&amp;').replace(/</g, '&lt;')
                .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
        };
        window.tooltipPbdHtml = function(text) {
            return window.escapePbdText(text).replace(/\r\n|\r|\n/g, '<br>');
        };

        window.get1CUrl = function(action, rawUrl, extraName) {
            var u = rawUrl ? btoa(unescape(encodeURIComponent(rawUrl))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '.') : "";
            var n = extraName ? btoa(unescape(encodeURIComponent(extraName))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '.') : "";
            return 'panelka://' + action + '?u=' + u + '&n=' + n;
        };

        /** НАВИГАЦИЯ С КЛАВИАТУРЫ (Стрелки + Переключение вкладок) */
        function initKeyboardNavigation() {
            document.addEventListener('keydown', function(e) {
                var keyCode = e.keyCode || e.which;
                
                var focused = document.activeElement;
                var menu = document.getElementById('context-menu');
                if (menu && menu.style.display === 'block' && menu.contains(focused)) {
                    if (keyCode === 27 || keyCode === 9) {
                        if (keyCode === 27) e.preventDefault();
                        closeContextMenu(true);
                        return;
                    }
                    if (keyCode === 32) {
                        e.preventDefault();
                        if (focused) focused.click();
                        return;
                    }
                    var items = Array.prototype.slice.call(menu.querySelectorAll('.c-menu-item'));
                    var itemIndex = items.indexOf(focused);
                    var nextItem = itemIndex;
                    if (keyCode === 40) nextItem = (itemIndex + 1) % items.length;
                    else if (keyCode === 38) nextItem = (itemIndex - 1 + items.length) % items.length;
                    else if (keyCode === 36) nextItem = 0;
                    else if (keyCode === 35) nextItem = items.length - 1;
                    else return;
                    e.preventDefault();
                    if (items[nextItem]) items[nextItem].focus();
                    return;
                }
                if (focused && focused.classList.contains('tab')) {
                    var visibleTabs = [];
                    for (var t = 0; t < pageNodes.length; t++) {
                        if (pageNodes[t].tab.style.display !== 'none') visibleTabs.push(pageNodes[t]);
                    }
                    var selectedIndex = -1;
                    for (var v = 0; v < visibleTabs.length; v++) {
                        if (visibleTabs[v].tab === focused) selectedIndex = v;
                    }
                    var nextTab = selectedIndex;
                    if (keyCode === 39) nextTab = (selectedIndex + 1) % visibleTabs.length;
                    else if (keyCode === 37) nextTab = (selectedIndex - 1 + visibleTabs.length) % visibleTabs.length;
                    else if (keyCode === 36) nextTab = 0;
                    else if (keyCode === 35) nextTab = visibleTabs.length - 1;
                    else return;
                    e.preventDefault();
                    if (visibleTabs[nextTab]) {
                        window.switchTab(visibleTabs[nextTab].id);
                        visibleTabs[nextTab].tab.focus();
                    }
                    return;
                }
                // Стандартная клавиша меню и Shift+F10 доступны для каждой ссылки.
                if (focused && focused.classList.contains('link-item') &&
                    (keyCode === 93 || (keyCode === 121 && e.shiftKey))) {
                    var focusedId = focused.id.match(/link_(\d+)/);
                    if (focusedId) window.showContextMenu(e, focused, parseInt(focusedId[1], 10));
                    return;
                }

                // Перехват Enter для запуска ссылки
                if (keyCode === 13) {
                    if (document.activeElement && document.activeElement.classList.contains('link-item')) {
                        e.preventDefault();
                        var linkIdMatch = document.activeElement.id.match(/link_(\d+)/);
                        if (linkIdMatch) {
                            var id = parseInt(linkIdMatch[1], 10);
                            var linkObj = linkRegistry[id];
                            if (linkObj && linkObj.Status === 'ОК') {
                                window.sendTo1C('open', linkObj.Url, linkObj.Name);
                            }
                        }
                        return;
                    }
                }

                if (keyCode < 37 || keyCode > 40) return; 
                
                var searchInput = document.getElementById('searchInput');
                var isSearchFocused = (document.activeElement === searchInput);
                var isLinkFocused = document.activeElement && document.activeElement.classList.contains('link-item');
                var isBodyFocused = (document.activeElement === document.body);

                if (isSearchFocused && (keyCode === 37 || keyCode === 39)) return;
                if (!isSearchFocused && !isLinkFocused && !isBodyFocused) return;

                var activeLinks = [];
                var allLinks = document.querySelectorAll('.tab-content.active .link-item');
                
                for(var i=0; i<allLinks.length; i++) {
                    var el = allLinks[i];
                    var card = el.closest('.card');
                    var row = el.closest('.link-row');
                    if (card && card.style.display !== 'none' && !card.classList.contains('collapsed') && row && !row.classList.contains('hidden')) {
                        activeLinks.push(el);
                    }
                }

                if (activeLinks.length === 0) return;

                if (keyCode === 40) { // Вниз
                    e.preventDefault();
                    if (isSearchFocused || isBodyFocused) activeLinks[0].focus();
                    else if (isLinkFocused) {
                        var idx = activeLinks.indexOf(document.activeElement);
                        if (idx !== -1 && idx < activeLinks.length - 1) activeLinks[idx + 1].focus();
                    }
                } else if (keyCode === 38) { // Вверх
                    e.preventDefault();
                    if (isLinkFocused) {
                        var idx = activeLinks.indexOf(document.activeElement);
                        if (idx > 0) activeLinks[idx - 1].focus();
                        else searchInput.focus();
                    }
                } else if (keyCode === 37 || keyCode === 39) { // Влево / Вправо
                    if (isLinkFocused) {
                        e.preventDefault();
                        var currentRect = document.activeElement.getBoundingClientRect();
                        var bestMatch = null;
                        var minDistance = Infinity;

                        for (var i = 0; i < activeLinks.length; i++) {
                            var cand = activeLinks[i];
                            if (cand === document.activeElement) continue;
                            
                            var rect = cand.getBoundingClientRect();
                            var dx = rect.left - currentRect.left;
                            var dy = rect.top - currentRect.top;

                            if (keyCode === 39 && dx > 10) { 
                                var dist = Math.abs(dy) * 5 + Math.abs(dx);
                                if (dist < minDistance) { minDistance = dist; bestMatch = cand; }
                            } else if (keyCode === 37 && dx < -10) { 
                                var dist = Math.abs(dy) * 5 + Math.abs(dx);
                                if (dist < minDistance) { minDistance = dist; bestMatch = cand; }
                            }
                        }
                        
                        if (bestMatch) {
                            bestMatch.focus();
                        } else {
                            switchTabByOffset(keyCode === 39 ? 1 : -1);
                        }
                    } else if (isBodyFocused) {
                        e.preventDefault();
                        switchTabByOffset(keyCode === 39 ? 1 : -1);
                    }
                }
            });
        }

        /** Перелистывание вкладок */
        function switchTabByOffset(offset) {
            var allTabs = document.querySelectorAll('.tab');
            var tabs = [];
            for (var t = 0; t < allTabs.length; t++) {
                if (allTabs[t].style.display !== 'none') tabs.push(allTabs[t]);
            }
            var activeTab = document.querySelector('.tab.active');
            if (tabs.length > 0 && activeTab) {
                var idx = tabs.indexOf(activeTab);
                var newIdx = idx + offset;
                if (idx >= 0 && newIdx >= 0 && newIdx < tabs.length) {
                    var targetTab = tabs[newIdx];
                    targetTab.click();
                    setTimeout(function() {
                        var newLinks = document.querySelectorAll('.tab-content.active .link-item');
                        for (var i = 0; i < newLinks.length; i++) {
                            var card = newLinks[i].closest('.card');
                            var row = newLinks[i].closest('.link-row');
                            if (card && card.style.display !== 'none' && !card.classList.contains('collapsed') && row && !row.classList.contains('hidden')) {
                                newLinks[i].focus();
                                return;
                            }
                        }
                        targetTab.focus();
                    }, 100);
                }
            }
        }

        window.switchTab = function(pageId, skipSearch, skipSave) {
            var selected = findPageNode(pageId);
            if (!selected) return;
            currentActiveTab = selected.id;
            if (!skipSave) window.sendTo1C('save_tab', selected.name);
            for (var i = 0; i < pageNodes.length; i++) {
                var page = pageNodes[i];
                var active = page === selected;
                page.tab.classList.toggle('active', active);
                page.tab.setAttribute('aria-selected', active ? 'true' : 'false');
                page.tab.tabIndex = active ? 0 : -1;
                page.content.classList.toggle('active', active);
            }
            if (!skipSearch) window.performSearch();
            else window.applyDynamicSpans();
        };

                if (!page.Columns || page.Columns.length === 0) {
                    var emptyTab = appendText(content, dashboardText('emptyTab'));
                    emptyTab.style.cssText = 'color:var(--text-muted); padding:20px;';
                } else {
                    for (var c = 0; c < page.Columns.length; c++) {
                        var colData = page.Columns[c];
                        var card = document.createElement('div');
                        if (!colData.Id) throw new Error('Не задан идентификатор колонки панели.');
                        var columnId = String(colData.Id);
                        var stateKey = JSON.stringify([pageId, columnId]);
                        var isCollapsed = collapsedCards[stateKey] === true;
                        card.setAttribute('data-state-key', stateKey);
                        card.className = 'card card-span-1' + (isCollapsed ? ' collapsed' : '');

                        var cardTitle = document.createElement('button');
                        cardTitle.type = 'button';
                        cardTitle.className = 'card-title';
                        cardTitle.setAttribute('aria-expanded', isCollapsed ? 'false' : 'true');
                        cardTitle.setAttribute('aria-controls', 'panelka_card_' + p + '_' + c);
                        cardTitle.onclick = function() { window.toggleCard(this); };
                        var caption = document.createElement('span');
                        caption.style.cssText = 'display:flex; align-items:center;';
                        appendIcon(caption, colData.IconBase64, 'col-icon');
                        var colName = appendText(caption, colData.Name, 'col-name-text');
                        colName.setAttribute('data-name', colData.Name);
                        colName.style.color = colData.Color;
                        cardTitle.appendChild(caption);
                        var collapseIcon = document.createElement('span');
                        collapseIcon.setAttribute('aria-hidden', 'true');
                        collapseIcon.innerHTML = collapseIconSvg;
                        cardTitle.appendChild(collapseIcon);
                        card.appendChild(cardTitle);

                        var cardBody = document.createElement('div'); 
                        cardBody.className = 'card-body';
                        cardBody.id = 'panelka_card_' + p + '_' + c;

                        if (colData.Types) {
                            for (var t2 = 0; t2 < colData.Types.length; t2++) {
                                var typeData2 = colData.Types[t2];
                                var group = document.createElement('div'); 
                                group.className = 'type-group';

                                if (typeData2.Name !== "Прочее" && DASHBOARD_DATA.Settings.GroupByType) {
                                    var typeHeader = document.createElement('div');
                                    typeHeader.className = 'type-header';
                                    appendIcon(typeHeader, typeData2.IconBase64);
                                    var typeName = appendText(typeHeader, typeData2.Name, 'type-name-text');
                                    typeName.setAttribute('data-name', typeData2.Name);
                                    group.appendChild(typeHeader);
                                }

                                if (typeData2.Links) {
                                    for (var l2 = 0; l2 < typeData2.Links.length; l2++) {
                                        var link = typeData2.Links[l2];
                                        link.globalId = linkRegistry.length; 
                                        linkRegistry.push(link);
                                        if (String(link.Url || '').indexOf('ext_file:') === 0) {
                                            if (!fileLinkRegistry[link.Url]) fileLinkRegistry[link.Url] = [];
                                            fileLinkRegistry[link.Url].push(link);
                                        }

                                        var row = document.createElement('div'); 
                                        row.className = 'link-row';

                                        var displayName = String(link.Name || '').replace(/[\n\r]/g, ' ');
                                        var displayPrefix = ''; 
                                        var errorText = ''; 
                                        var urlInfo = isAdmin ? '\n' + link.Url : '';
                                        
                                        if (link.Status === 'НетПрав') { 
                                            displayPrefix = dashboardText('noRightsPrefix'); 
                                            errorText = dashboardText('noRights') + urlInfo; 
                                        } else if (link.Status === 'Удален') { 
                                            displayPrefix = dashboardText('deletedPrefix'); 
                                            errorText = dashboardText('deleted') + urlInfo; 
                                        } else if (link.Status === 'НеНайден') { 
                                            displayPrefix = dashboardText('missingPrefix'); 
                                            errorText = dashboardText('missing') + urlInfo; 
                                        } else if (link.Status === 'Недоступен') {
                                            displayPrefix = dashboardText('unavailablePrefix');
                                            errorText = dashboardText('unavailable') + urlInfo;
                                        } else if (link.Status === 'Ошибка') { 
                                            displayPrefix = dashboardText('errorPrefix'); 
                                            errorText = dashboardText('checkFailed') + urlInfo; 
                                        }
                                        
                                        displayName = displayPrefix + displayName;
                                        var container = document.createElement('div');
                                        container.className = 'link-container';
                                        var header = document.createElement('div');
                                        header.className = 'link-header';
                                        var linkEl = document.createElement('a');
                                        linkEl.tabIndex = 0;
                                        linkEl.id = 'link_' + link.globalId;
                                        linkEl.className = 'link-item' + (link.Status === 'ОК' ? '' : ' link-error');
                                        linkEl.setAttribute('data-name', displayName);
                                        linkEl.setAttribute('data-raw-url', link.Url);
                                        if (link.Status === 'ОК') {
                                            linkEl.href = window.get1CUrl('open', link.Url);
                                            linkEl.target = '1c_bridge';
                                        } else {
                                            linkEl.setAttribute('aria-disabled', 'true');
                                            linkEl.onclick = function() { return false; };
                                        }
                                        renderHighlightedText(linkEl, displayName, '', link.IsFavorite ? ' ⭐' : '');
                                        header.appendChild(linkEl);
                                        container.appendChild(header);
                                        if (link.Tooltip) {
                                            var currentTMode = tMode;
                                            if (link.Status !== 'ОК' && currentTMode === 'Нет') currentTMode = 'Снизу';
                                            if (currentTMode === 'Всплывающая') linkEl.title = link.Tooltip;
                                            else if (currentTMode === 'Снизу' || currentTMode === 'Кнопка') {
                                                var tooltip = document.createElement('div');
                                                tooltip.className = 'tooltip' + (currentTMode === 'Кнопка' ? ' hidden-tooltip' : '');
                                                tooltip.textContent = link.Tooltip;
                                                tooltip.id = 'panelka_tooltip_' + link.globalId;
                                                container.appendChild(tooltip);
                                                if (currentTMode === 'Кнопка') {
                                                    var helpButton = document.createElement('button');
                                                    helpButton.type = 'button';
                                                    helpButton.className = 'help-btn';
                                                    helpButton.textContent = '?';
                                                    helpButton.setAttribute('aria-label', dashboardText('helpPrefix') + displayName);
                                                    helpButton.setAttribute('aria-controls', tooltip.id);
                                                    helpButton.setAttribute('aria-expanded', 'false');
                                                    helpButton.onclick = function(event) { window.toggleTooltip(event, this); };
                                                    header.appendChild(helpButton);

                                            }
                                        }
                                        if (link.Status !== 'ОК') appendText(container, errorText, 'error-text');
                                        row.appendChild(container);
                                        var dropdown = document.createElement('div');
                                        dropdown.className = 'dropdown';
                                        var menuButton = document.createElement('button');
                                        menuButton.type = 'button';
                                        menuButton.className = 'menu-btn';
                                        menuButton.textContent = '⋮';
                                        menuButton.setAttribute('aria-label', dashboardText('actionsPrefix') + displayName);
                                        menuButton.setAttribute('aria-haspopup', 'menu');
                                        menuButton.setAttribute('aria-controls', 'context-menu');
                                        menuButton.setAttribute('aria-expanded', 'false');

                                            return function(event) { window.showContextMenu(event, this, id); };
                                        })(link.globalId);
                                        dropdown.appendChild(menuButton);
                                        row.appendChild(dropdown);

                                        row.oncontextmenu = (function(id) { 
                                            return function(e) { window.showContextMenu(e, null, id); };

                                if (group.childNodes.length > 0) cardBody.appendChild(group);

                        card.appendChild(cardBody);
                        content.appendChild(card);
                    }
                }
                dashContainer.appendChild(content);
            }
            fixInlineColorsForDarkMode();
        }

        window.applyDynamicSpans = function() {
            var containerWidth = document.getElementById('dashboard').clientWidth || window.innerWidth;
            var maxGridCols = Math.floor((containerWidth + 25) / 365);
            if (maxGridCols < 1) maxGridCols = 1;
            
            var vh = window.innerHeight || document.documentElement.clientHeight || 800;
            var availableHeight = vh - 180; 
            var rowHeight = 30; 
            var maxItemsPerColumn = Math.floor(availableHeight / rowHeight);
            
            if (maxItemsPerColumn < 8) maxItemsPerColumn = 8;   
            if (maxItemsPerColumn > 15) maxItemsPerColumn = 15; 
            
            var activeContent = document.querySelector('.tab-content.active');
            if (!activeContent) return;

            var cards = activeContent.querySelectorAll('.card');
            cards.forEach(function(card) {
                if (card.style.display === 'none') return;
                
                var weight = 0;
                card.querySelectorAll('.link-row:not(.hidden)').forEach(function() { weight += 1; });
                card.querySelectorAll('.type-header:not(.hidden)').forEach(function() { weight += 1.5; });
                
                var span = Math.ceil(weight / maxItemsPerColumn);
                if (span < 1) span = 1; 
                if (span > maxGridCols) span = maxGridCols; 
                
                card.className = card.className.replace(/card-span-\d+/g, '').trim();
                card.classList.add('card-span-' + span);
            });
        };

        function fixInlineColorsForDarkMode() {
            var isDark = document.body.className.indexOf("dark-mode") > -1;
            var spans = document.querySelectorAll('.col-name-text');
            for(var i = 0; i < spans.length; i++) {
                if(isDark) { 
                    spans[i].setAttribute('data-orig-color', spans[i].style.color); 
                    spans[i].style.color = '#FFFFFF'; 
                } else { 
                    var origColor = spans[i].getAttribute('data-orig-color'); 
                    if(origColor) spans[i].style.color = origColor; 
                }
            }
        }

        window.toggleTooltip = function(e, btn) { 
            e.preventDefault(); 
            e.stopPropagation(); 
            var tt = btn.parentElement.parentElement.querySelector('.hidden-tooltip'); 
            if (tt) {
                var open = tt.style.display !== 'block';
                tt.style.display = open ? 'block' : 'none';
                btn.setAttribute('aria-expanded', open ? 'true' : 'false');
            } 
        };

        window.showContextMenu = function(e, btnNode, id) {
            e.preventDefault(); 
            e.stopPropagation();
            
            var menu = document.getElementById('context-menu'); 
            if (!menu) return;
            
            if (menu.style.display === 'block' && window.currentMenuLinkId === id) {
                closeContextMenu(true);
                return false;
            }
            closeContextMenu(false);
            contextMenuTrigger = btnNode || (e.currentTarget && e.currentTarget.querySelector('.menu-btn')) || document.getElementById('link_' + id);
            if (contextMenuTrigger) contextMenuTrigger.setAttribute('aria-expanded', 'true');
            window.currentMenuLinkId = id;
            
            menu.innerHTML = '';
            var link = linkRegistry[id];
            if (!link) { closeContextMenu(false); return false; }

            function addItem(text, action, stringToCopy) { 
                var el = document.createElement('a'); 
                el.className = 'c-menu-item'; 
                el.textContent = text;
                el.setAttribute('role', 'menuitem');
                el.tabIndex = -1; 
                el.target = '1c_bridge';
                el.href = window.get1CUrl(action, link.Url, stringToCopy || "");
                
                el.onclick = function(event) {
                    event.preventDefault();
                    event.stopPropagation();
                    closeContextMenu(true);
                    // Отдельная ссылка моста передаёт событие 1С, независимо от скрытия меню.
                    // Исходный переход отменён: команда отправляется ровно один раз.
                    window.sendTo1C(action, link.Url, stringToCopy || "");
                    return false;
                }; 
                menu.appendChild(el); 
            }

            if (link.Status === 'ОК') {
                addItem(dashboardText('open'), 'open');
                if (link.IsSysCmd || link.IsExtFile || link.IsAddProc || (link.IsStandard && !link.IsDataLink && !link.IsWebLink)) { 
                    addItem(dashboardText(link.IsAddProc || link.IsExtFile ? 'runNew' : 'openNew'), 'open_new'); 
                }
            }

            var canFav = false; 
            if (link.Status === 'ОК' && ((link.IsAddProc && DASHBOARD_DATA.Settings.IsAdmin) || link.IsStandard)) { 
                canFav = true; 
            }

            if ((link.IsAddProc && DASHBOARD_DATA.Settings.IsAdmin && link.Status === 'ОК') || canFav || link.IsFavorite) {
                var divider = document.createElement('div');
                divider.className = 'dropdown-divider';
                menu.appendChild(divider);
                
                if (link.IsAddProc && DASHBOARD_DATA.Settings.IsAdmin && link.Status === 'ОК') { 
                    addItem(dashboardText('openCard'), 'open_card'); 
                }
                if (canFav || link.IsFavorite) { 
                    var favAddText = dashboardText(link.IsAddProc ? 'addObjectFavorite' : 'addFavorite');
                    var favRemoveText = dashboardText(link.IsAddProc ? 'removeObjectFavorite' : 'removeFavorite');
                    if (link.IsFavorite) addItem(favRemoveText, 'remove_fav', link.Name); 
                    else addItem(favAddText, 'add_fav', link.Name); 
                }
            }

            var divider = document.createElement('div');
                divider.className = 'dropdown-divider';
                menu.appendChild(divider);
            addItem(dashboardText('copyName'), 'copy_text', link.Name);

            if (link.Status === 'ОК') {
                if (link.IsWebLink) { 
                    addItem(dashboardText('copyWeb'), 'copy_url', link.Url); 
                } else if (link.IsExtFile) { 
                    addItem(dashboardText('copyFile'), 'copy_file', link.Url); 
                    addItem(dashboardText('copyDirectory'), 'copy_dir', link.Url); 
                } else if (link.IsAddProc) { 
                    if (DASHBOARD_DATA.Settings.IsAdmin) { 
                        addItem(dashboardText('copyObjectUrl'), 'copy_text', link.Url); 
                    } 
                } else if (link.IsStandard) { 
                    addItem(dashboardText('copyInternalUrl'), 'copy_text', link.Url); 
                    addItem(dashboardText('copyExternalUrl'), 'copy_ext_url', link.Url); 
                }
            }

            if (link.HasTooltip) addItem(dashboardText('copyTooltip'), 'copy_text', link.Tooltip);

            menu.style.display = 'block';
            var topPos = e.clientY; 
            var leftPos = e.clientX;
            
            if(btnNode) { 
                var rect = btnNode.getBoundingClientRect(); 
                topPos = rect.bottom; 
                leftPos = rect.left - menu.offsetWidth + 20; 
            }
            if (leftPos < 10) leftPos = 10;
            if (topPos + menu.offsetHeight > window.innerHeight) topPos = window.innerHeight - menu.offsetHeight - 10;
            if (!isFinite(topPos) || topPos < 10) topPos = 10;
            if (!isFinite(leftPos) || leftPos < 10) leftPos = 10;
            
            menu.style.top = topPos + 'px'; 
            menu.style.left = leftPos + 'px';
            var firstItem = menu.querySelector('.c-menu-item');
            if (firstItem) firstItem.focus();
            return false;
        };

        window.clearSearch = function() { 
            var input = document.getElementById('searchInput'); 
            if (input) { 
                input.value = ''; 
                window.performSearch(); 
                input.focus(); 
            } 
        };
        
        window.applySearchFrom1C = function(searchText) { 
            var input = document.getElementById('searchInput'); 
            if (input) { input.value = searchText; } 
            window.performSearch(); 
        };

        window.performSearch = function() {
            var input = document.getElementById('searchInput'); 
            if (!input) return;
            
            var filter = input.value.toLowerCase();
            var clearBtn = document.getElementById('searchClear');
            if (clearBtn) clearBtn.style.display = (filter.length > 0) ? 'block' : 'none';
            
            var isSearching = filter.length > 0;
            var firstVisibleTab = null; 
            var tabContents = document.querySelectorAll('.tab-content');
            var collapsedCards = readCollapsedCards();

            for (var t = 0; t < tabContents.length; t++) {
                var content = tabContents[t]; 
                var tabHasVisibleCards = false; 
                var tabName = content.getAttribute('data-tab');
                var cards = content.querySelectorAll('.card');
                
                for (var c = 0; c < cards.length; c++) {
                    var card = cards[c]; 
                    var cardHasVisibleLinks = false;
                    var colNameSpan = card.querySelector('.col-name-text');
                    var origColName = colNameSpan ? colNameSpan.getAttribute('data-name') : "";
                    var colMatch = filter !== "" && origColName.toLowerCase().indexOf(filter) > -1;

                    renderHighlightedText(colNameSpan, origColName, filter);

                    var typeGroups = card.querySelectorAll('.type-group');
                    for (var g = 0; g < typeGroups.length; g++) {
                        var group = typeGroups[g]; 
                        var groupHasVisibleLinks = false;
                        var typeNameSpan = group.querySelector('.type-name-text');
                        var origTypeName = typeNameSpan ? typeNameSpan.getAttribute('data-name') : "";
                        var typeMatch = filter !== "" && origTypeName.toLowerCase().indexOf(filter) > -1;

                        renderHighlightedText(typeNameSpan, origTypeName, filter);

                        var rows = group.querySelectorAll('.link-row');
                        for (var r = 0; r < rows.length; r++) {
                            var row = rows[r]; 
                            var linkItem = row.querySelector('.link-item'); 
                            if (!linkItem) continue;
                            
                            var originalText = linkItem.getAttribute('data-name');
                            var linkIdMatch = linkItem.id.match(/link_(\d+)/);
                            var linkId = linkIdMatch ? parseInt(linkIdMatch[1], 10) : -1;
                            var fObj = linkRegistry[linkId];
                            var favIco = (fObj && fObj.IsFavorite) ? ' ⭐' : '';
                            
                            if (filter === "") {
                                renderHighlightedText(linkItem, originalText, '', favIco); 
                                row.classList.remove('hidden'); 
                                groupHasVisibleLinks = true;
                            } else {
                                var linkMatch = originalText.toLowerCase().indexOf(filter) > -1;
                                if (linkMatch) {
                                    renderHighlightedText(linkItem, originalText, filter, favIco);
                                    row.classList.remove('hidden'); 
                                    groupHasVisibleLinks = true;
                                } else if (colMatch || typeMatch) {
                                    renderHighlightedText(linkItem, originalText, '', favIco);
                                    row.classList.remove('hidden'); 
                                    groupHasVisibleLinks = true;
                                } else { 
                                    row.classList.add('hidden'); 
                                }
                            }
                        }
                        var header = group.querySelector('.type-header');
                        if (header) { 
                            if (filter !== "" && !groupHasVisibleLinks) header.classList.add('hidden'); 
                            else header.classList.remove('hidden'); 
                        }
                        
                        if (groupHasVisibleLinks || filter === "") group.style.display = 'block'; 
                        else group.style.display = 'none';
                        
                        if (groupHasVisibleLinks) cardHasVisibleLinks = true;
                    }
                    if (cardHasVisibleLinks || filter === "") { 
                        card.style.display = 'inline-block'; 
                        tabHasVisibleCards = true; 
                        
                        setCardCollapsed(card, !isSearching && collapsedCards[card.getAttribute('data-state-key')] === true);
                    } else { 
                        card.style.display = 'none'; 
                    }
                }

                var pageNode = findPageNode(tabName);
                var tabHeader = pageNode ? pageNode.tab : null;
                if (tabHeader) {
                    if (filter === "" || tabHasVisibleCards) { 
                        tabHeader.style.display = 'flex'; 
                        if (!firstVisibleTab) firstVisibleTab = tabName; 
                    } else { 
                        tabHeader.style.display = 'none'; 
                    }
                }
            }
            if (filter !== "") { 
                var activeTabHeader = document.querySelector('.tab.active'); 
                if (activeTabHeader && activeTabHeader.style.display === 'none' && firstVisibleTab) { 
                    window.switchTab(firstVisibleTab, true, true); 
                } 
            } else { 
                if (currentActiveTab && findPageNode(currentActiveTab)) { 
                    window.switchTab(currentActiveTab, true, true); 
                } 
            }
            window.applyDynamicSpans();
        };

        window.onload = window.init;
